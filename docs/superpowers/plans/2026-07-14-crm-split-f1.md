import { PageHeader } from '../../components/PageHeader'

import { Megaphone, ChatCircleDots, ChartBar } from '@phosphor-icons/react'
import { CrmPipelineTab } from './CrmPipelineTab'

type CrmTab = 'pipeline' | 'whatsapp' | 'estadisticas'

`backend/app/Http/Controllers/Api/CrmEstadisticasController.php`:
```php
<?php

namespace App\Http\Controllers\Api;

use App\Models\Usuario;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\DB;

class CrmEstadisticasController extends Controller
{
    public function resumen(Request $request): JsonResponse
    {
        $user = Auth::user();
        $tiendaId = trim((string) $request->query('tienda_id'));

        $esAdmin = $user->tieneAlgunRol(Usuario::ROL_ADMINISTRADOR, Usuario::ROL_GERENTE);
        if (!$esAdmin && $tiendaId !== trim((string) $user->tienda_id)) {
            return response()->json(['message' => 'No tienes permisos sobre esta tienda.'], 403);
        }

        $leadsActivos = DB::table('crm_leads')
            ->where('tienda_id', $tiendaId)
            ->whereNotIn('estado', ['GANADO', 'PERDIDO'])
            ->count();

        $inicioMes = now()->startOfMonth();
        $totalLeadsMes = DB::table('crm_leads')
            ->where('tienda_id', $tiendaId)
            ->where('created_at', '>=', $inicioMes)
            ->count();
        $ganadosMes = DB::table('crm_leads')
            ->where('tienda_id', $tiendaId)
            ->where('estado', 'GANADO')
            ->where('updated_at', '>=', $inicioMes)
            ->count();
        $conversionMes = $totalLeadsMes > 0 ? round(($ganadosMes / $totalLeadsMes) * 100, 1) : 0.0;

        $ventasMes = (float) DB::table('reportes as r')
            ->join('reporte_categorias as rc', 'rc.reporte_id', '=', 'r.id')
            ->where('r.tienda_id', $tiendaId)
            ->where('r.fecha', '>=', $inicioMes->toDateString())
            ->sum('rc.total');

        return response()->json([
            'leads_activos'  => $leadsActivos,
            'conversion_mes' => $conversionMes,
            'ventas_mes'     => $ventasMes,
        ]);
    }
}
```
