{
    $request->validate(['dni' => 'required|string', 'huella' => 'required|string']);

    $agente = Agente::where('dni', $request->dni)->where('activo', true)->firstOrFail();
    $fecha  = now()->toDateString();

        ->where('agente_id', $agente->id)
        ->whereDate('fecha', $fecha)
        ->first();

    \DB::table('asistencias')
        ->where('id', $asistencia->id)
        ->update([
            'omitio_refrigerio' => true,
            'siguiente_marcacion' => 'salida',
            'updated_at' => now(),
        ]);

    return response()->json([
        'message'           => 'Turno corrido activado. Su próxima marcación es la salida.',

```php
public function liquidacionAsistencias(Request $request, int $id): JsonResponse
{
    abort_if(auth()->user()->rol !== 'admin', 403);

    $agente = \App\Models\Agente::findOrFail($id);
    $mes    = $request->input('mes', now()->format('Y-m')); // e.g. "2026-06"
    [$y, $m] = explode('-', $mes);

    $asistencias = \DB::table('asistencias')
        ->where('agente_id', $id)
        ->whereYear('fecha', $y)
        ->whereMonth('fecha', $m)
        ->orderBy('fecha')
        ->get();

    $valorMinuto = (float) config('attendance.valor_minuto_tardanza', 0.10); // S/ por minuto

    $dias = $asistencias->map(function ($a) use ($valorMinuto) {
        $tardanza       = (int) ($a->minutos_tardanza ?? 0);
        $deuda          = (int) ($a->minutos_deuda ?? 0);
        $comodin        = (bool) ($a->uso_comodin ?? false);
        $omitioRefri    = (bool) ($a->omitio_refrigerio ?? false);
        $descuento      = $comodin ? 0 : ($deuda * $valorMinuto);
        return [
            'fecha'             => $a->fecha,
            'estado'            => $a->estado_asistencia ?? 'presente',
            'hora_entrada'      => $a->hora_entrada,
            'hora_salida'       => $a->hora_salida,
            'minutos_tardanza'  => $tardanza,
            'minutos_deuda'     => $deuda,
            'uso_comodin'       => $comodin,
            'omitio_refrigerio' => $omitioRefri,
            'descuento_soles'   => round($descuento, 2),
        ];
    });

    $resumen = [
        'total_tardanzas_min'   => $dias->sum('minutos_tardanza'),
        'deuda_acumulada_min'   => $dias->sum('minutos_deuda'),
        'comodines_usados'      => $dias->where('uso_comodin', true)->count(),
        'total_descuento_soles' => round($dias->sum('descuento_soles'), 2),
    ];

    return response()->json([
        'agente'  => ['id' => $agente->id, 'nombre' => $agente->nombre, 'dni' => $agente->dni],
        'mes'     => $mes,
        'dias'    => $dias,
        'resumen' => $resumen,
    ]);
}
```

import { useState } from 'react'
import { useQuery } from '@tanstack/react-query'
import { api } from '../../services/api'

export function HistorialLiquidacionPage() {
  const [agenteId, setAgenteId] = useState('')
  const [mes, setMes] = useState(() => new Date().toISOString().slice(0, 7))

  const { data, isLoading } = useQuery({
    queryKey: ['liquidacion', agenteId, mes],
    queryFn: () => api.get(`/agentes/${agenteId}/liquidacion-asistencias?mes=${mes}`).then(r => r.data),
    enabled: !!agenteId,
  })

  return (
    <div className="mx-auto max-w-4xl space-y-6 p-6">
      <h1 className="text-xl font-bold text-zinc-100">Liquidación de Asistencias</h1>

      <div className="flex gap-3">
        <input
          type="number"
          placeholder="ID Agente"
          value={agenteId}
          onChange={e => setAgenteId(e.target.value)}
          className="w-32 rounded-lg bg-zinc-800 px-3 py-2 text-sm text-zinc-100"
        />
        <input
          type="month"
          value={mes}
          onChange={e => setMes(e.target.value)}
          className="rounded-lg bg-zinc-800 px-3 py-2 text-sm text-zinc-100"
        />
      </div>

      {isLoading && <p className="text-zinc-400">Cargando...</p>}

      {data && (
        <>
          <div className="rounded-xl border border-zinc-700 bg-zinc-900 p-4">
            <p className="font-semibold text-zinc-100">{data.agente.nombre} — DNI {data.agente.dni}</p>
            <p className="text-sm text-zinc-400">{data.mes}</p>
          </div>

          {/* Resumen */}
          <div className="grid grid-cols-2 gap-3 sm:grid-cols-4">
            {[
              { label: 'Tardanzas (min)', value: data.resumen.total_tardanzas_min },
              { label: 'Deuda acum. (min)', value: data.resumen.deuda_acumulada_min },
              { label: 'Comodines usados', value: data.resumen.comodines_usados },
              { label: 'Descuento (S/)', value: `S/ ${data.resumen.total_descuento_soles}` },
            ].map(({ label, value }) => (
              <div key={label} className="rounded-xl border border-zinc-700 bg-zinc-800 p-3 text-center">
                <p className="text-xl font-bold text-zinc-100">{value}</p>
                <p className="text-xs text-zinc-400">{label}</p>
              </div>
            ))}
          </div>

          {/* Tabla de días */}
          <table className="w-full text-sm">
            <thead>
              <tr className="border-b border-zinc-700 text-left text-xs text-zinc-400">
                <th className="pb-2">Fecha</th>
                <th className="pb-2">Estado</th>
                <th className="pb-2">Entrada</th>
                <th className="pb-2">Salida</th>
                <th className="pb-2">Tardanza</th>
                <th className="pb-2">Deuda</th>
                <th className="pb-2">Comodín</th>
                <th className="pb-2">Descuento</th>
              </tr>
            </thead>
            <tbody>
              {data.dias.map((d: any) => (
                <tr key={d.fecha} className="border-b border-zinc-800 text-zinc-300">
                  <td className="py-1.5">{d.fecha}</td>
                  <td className="py-1.5 capitalize">{d.estado}</td>
                  <td className="py-1.5">{d.hora_entrada ?? '—'}</td>
                  <td className="py-1.5">{d.hora_salida ?? '—'}</td>

}
